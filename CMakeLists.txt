cmake_minimum_required(VERSION 3.16)
if(POLICY CMP0141)
    cmake_policy(SET CMP0141 NEW)
    set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "$<$<CONFIG:Debug,RelWithDebInfo>:Embedded>")
endif()
project(RaceCar3D VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(USE_MSVC_RUNTIME_LIBRARY_DLL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/glfw-3.5.1)
add_library(glad STATIC third_party/glad/src/glad.c)
target_include_directories(glad PUBLIC third_party/glad/include)
add_executable(RaceCar3D Main.cpp src/Circuit.cpp src/Race.cpp src/Renderer.cpp src/Hud.cpp)
target_include_directories(RaceCar3D PRIVATE third_party/glm)
target_link_libraries(RaceCar3D PRIVATE glad glfw)
target_compile_definitions(RaceCar3D PRIVATE APEX_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" NOMINMAX)
if(WIN32)
    target_link_libraries(RaceCar3D PRIVATE opengl32)
endif()
if(MSVC)
    target_compile_options(RaceCar3D PRIVATE /W3 /permissive- /utf-8)
else()
    target_compile_options(RaceCar3D PRIVATE -Wall -Wextra)
endif()
add_custom_command(TARGET RaceCar3D POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Shaders" "$<TARGET_FILE_DIR:RaceCar3D>/Shaders"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Assets" "$<TARGET_FILE_DIR:RaceCar3D>/Assets")
enable_testing()
add_test(NAME RaceChoreographyAndAssets COMMAND RaceCar3D --validate)
